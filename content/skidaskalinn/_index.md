const app = getApps().length ? getApp() : initializeApp(cfg);
const db = getFirestore(app);

const IS_MONTHS = ['Janúar','Febrúar','Mars','Apríl','Maí','Júní','Júlí','Ágúst','September','Október','Nóvember','Desember'];
const IS_DAYS   = ['Mán','Þri','Mið','Fim','Fös','Lau','Sun'];

let bookedDates = new Set();
let currentYear, currentMonth;

function toKey(d) {
  return d.getFullYear() + '-' + String(d.getMonth()+1).padStart(2,'0') + '-' + String(d.getDate()).padStart(2,'0');
}

function renderCal() {
  const grid = document.getElementById('cal-grid');
  document.getElementById('cal-month-label').textContent = IS_MONTHS[currentMonth] + ' ' + currentYear;
  const today = toKey(new Date());
  const firstDay = new Date(currentYear, currentMonth, 1);
  const daysInMonth = new Date(currentYear, currentMonth+1, 0).getDate();
  let startDow = firstDay.getDay(); // 0=Sun
  startDow = startDow === 0 ? 6 : startDow - 1; // Mon=0

  let html = IS_DAYS.map(d => `<div class="cal-day-header">${d}</div>`).join('');
  for (let i = 0; i < startDow; i++) html += `<div class="cal-day empty"></div>`;
  for (let d = 1; d <= daysInMonth; d++) {
    const key = currentYear + '-' + String(currentMonth+1).padStart(2,'0') + '-' + String(d).padStart(2,'0');
    const cls = bookedDates.has(key) ? 'booked' : 'free';
    const todayCls = key === today ? ' today' : '';
    html += `<div class="cal-day ${cls}${todayCls}">${d}</div>`;
  }
  grid.innerHTML = html;
}

window.calPrev = function() { currentMonth--; if (currentMonth < 0) { currentMonth=11; currentYear--; } renderCal(); };
window.calNext = function() { currentMonth++; if (currentMonth > 11) { currentMonth=0; currentYear++; } renderCal(); };

async function init() {
  const now = new Date();
  currentYear = now.getFullYear();
  currentMonth = now.getMonth();
  try {
    const snap = await getDocs(collection(db, 'skidaskalinn_bookings'));
    snap.forEach(doc => {
      const { start, end } = doc.data();
      const s = new Date(start), e = new Date(end);
      for (let d = new Date(s); d <= e; d.setDate(d.getDate()+1)) {
        bookedDates.add(toKey(new Date(d)));
      }
    });
  } catch(err) { console.error('Calendar error:', err); }
  renderCal();
}
init();

## Ski Cabin Skagaströnd — Accommodation in Iceland

The Ski Cabin (*Skíðaskálinn*) is located on the slopes of Mt. Spákonufell, just outside the village of Skagaströnd in North Iceland. It sleeps up to 23 guests (8 beds + sleeping bag space for approx. 15) and is open year-round.

**Facilities:** Full kitchen · 2 bathrooms · Natural spring water · Parking on site

Skagaströnd is within easy reach of the Vatnsnes peninsula (seal watching), Kolugljúfur canyon, and the Skagafjörður fjord — a great base for hiking, bird watching, fishing, and exploring North Iceland.
